cmake_minimum_required(VERSION 3.20)
project(FM26_AI_MANAGER LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(FM26_AI_MANAGER WIN32 FM26_AI_MANAGER.cpp)
target_compile_definitions(FM26_AI_MANAGER PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(FM26_AI_MANAGER PRIVATE /W4 /EHsc /utf-8)
  set_property(TARGET FM26_AI_MANAGER PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
