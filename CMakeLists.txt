cmake_minimum_required(VERSION 3.20)
project(FM26_AI_MANAGER LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(FM26_AI_MANAGER WIN32 FM26_AI_MANAGER.cpp)
target_compile_definitions(FM26_AI_MANAGER PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(FM26_AI_MANAGER PRIVATE user32 gdi32 msimg32)
if(MSVC)
  target_compile_options(FM26_AI_MANAGER PRIVATE /W4 /permissive- /EHsc /utf-8)
endif()
